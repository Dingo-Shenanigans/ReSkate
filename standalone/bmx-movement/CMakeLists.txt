cmake_minimum_required(VERSION 3.20)
project(bmx_movement VERSION 0.3.0 LANGUAGES CXX)

option(BMX_BUILD_DEMO "Build the Windows OpenGL demo" ON)
option(BMX_BUILD_TESTS "Build standalone movement tests" ON)

add_library(bmx_movement STATIC src/movement.cpp src/grind.cpp src/suspension.cpp src/attitude.cpp src/joint_actuation.cpp src/observed_joint_actuation_tracks.cpp src/observed_pose_tracks.cpp src/observed_vehicle_pose.cpp src/observed_ik_tracks.cpp)
add_library(bmx::movement ALIAS bmx_movement)
set_target_properties(bmx_movement PROPERTIES EXPORT_NAME movement)
target_include_directories(bmx_movement PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(bmx_movement PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(bmx_movement PRIVATE /W4 /permissive-)
else()
  target_compile_options(bmx_movement PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BMX_BUILD_DEMO AND WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/demo/demo_win32.cpp")
  add_executable(bmx_demo WIN32 demo/demo_win32.cpp)
  target_link_libraries(bmx_demo PRIVATE bmx_movement opengl32 gdi32 user32)
  target_compile_definitions(bmx_demo PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MINGW)
    target_link_options(bmx_demo PRIVATE -static)
  endif()
endif()

if(BMX_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/tests.cpp")
  enable_testing()
  add_executable(bmx_tests tests/tests.cpp)
  target_link_libraries(bmx_tests PRIVATE bmx_movement)
  if(MINGW)
    target_link_options(bmx_tests PRIVATE -static)
  endif()
  add_test(NAME bmx_movement COMMAND bmx_tests)
  add_executable(bmx_pose_tests tests/pose_tracks.cpp)
  target_link_libraries(bmx_pose_tests PRIVATE bmx_movement)
  if(MINGW)
    target_link_options(bmx_pose_tests PRIVATE -static)
  endif()
  add_test(NAME bmx_pose_tracks COMMAND bmx_pose_tests)
  add_executable(bmx_joint_actuation_tests tests/joint_actuation.cpp)
  target_link_libraries(bmx_joint_actuation_tests PRIVATE bmx_movement)
  if(MINGW)
    target_link_options(bmx_joint_actuation_tests PRIVATE -static)
  endif()
  add_test(NAME bmx_joint_actuation COMMAND bmx_joint_actuation_tests)
  add_executable(bmx_vehicle_pose_tests tests/vehicle_pose.cpp)
  target_link_libraries(bmx_vehicle_pose_tests PRIVATE bmx_movement)
  if(MINGW)
    target_link_options(bmx_vehicle_pose_tests PRIVATE -static)
  endif()
  add_test(NAME bmx_vehicle_pose COMMAND bmx_vehicle_pose_tests)
  add_executable(bmx_input_holdpoint_tests tests/input_holdpoints.cpp)
  target_link_libraries(bmx_input_holdpoint_tests PRIVATE bmx_movement)
  if(MINGW)
    target_link_options(bmx_input_holdpoint_tests PRIVATE -static)
  endif()
  add_test(NAME bmx_input_holdpoints COMMAND bmx_input_holdpoint_tests)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bmx_movementConfig.cmake"
  "include(\"\${CMAKE_CURRENT_LIST_DIR}/bmx_movement-targets.cmake\")\n")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/bmx_movementConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS bmx_movement EXPORT bmx_movement_targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT bmx_movement_targets FILE bmx_movement-targets.cmake NAMESPACE bmx:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bmx_movement)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/bmx_movementConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/bmx_movementConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bmx_movement)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/bmx_movement)
